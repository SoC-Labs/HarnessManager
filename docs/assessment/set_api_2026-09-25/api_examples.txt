# SET-API: example /settings replies from the real daemon app (create_app over the demo engine, TestClient),
# a scratch state dir and policy.toml under /tmp, the service's env with $HARNESS_MANAGER_OPENOCD set.
# The script, then its output (the secret sent was tok-SECRET; it appears nowhere below).

```python
import json, warnings
from pathlib import Path
with warnings.catch_warnings():
    warnings.simplefilter("ignore")
    from fastapi.testclient import TestClient
from harness_manager.daemon.app import create_app
from harness_manager.daemon.hosts import allowed_hosts
from harness_manager.demo import DemoEngine

root = Path("/tmp/setapi-demo/api"); root.mkdir(parents=True, exist_ok=True)
pol = root / "policy.toml"
pol.write_text('[lock]\ntools.vivado = "/tools/Xilinx/Vivado/2024.1/bin/vivado"\n')
H = {"Authorization": "Bearer demo"}
eng = DemoEngine(speed=0)
app = create_app(eng, token="demo", state_dir=root / "state", static_dir=None,
                 allowed_hosts=allowed_hosts("127.0.0.1"))
d = app.state.daemon
d.settings.policy_path = pol
d.settings.env = {"HARNESS_MANAGER_KEYRING": "off", "HARNESS_MANAGER_OPENOCD": "/env/openocd"}
events = []
d.bus.subscribe("settings.changed", lambda e: events.append(e.data))
def show(title, r, trim=None):
    body = r.json()
    if trim: body = trim(body)
    print(f"### {title}\nHTTP {r.status_code}\n{json.dumps(body, indent=1)}\n")
with TestClient(app, base_url="http://127.0.0.1:40123") as c:
    show("PUT /settings (all or nothing; apply class)", c.put("/api/v1/settings", headers=H,
         json={"tools.openocd": "/opt/openocd/bin/openocd", "hubs.lab.host": "mapstone-dev.ecs.soton.ac.uk", "advanced.port": 0}),
         lambda b: {**b, "rows": [{k: r[k] for k in ("key","value","source","where","shadowed","apply")} for r in b["rows"]]})
    show("GET /settings?key=tools.openocd (env shadowing, the service's env)", c.get("/api/v1/settings?key=tools.openocd", headers=H),
         lambda b: {"files": b["files"], "policy": b["policy"], "rows": b["rows"]})
    show("PUT /settings with a locked key -> 409 (nothing written)", c.put("/api/v1/settings", headers=H, json={"general.theme": "dark", "tools.vivado": "/mine"}))
    show("PUT /settings with one bad value -> 400 (nothing written)", c.put("/api/v1/settings", headers=H, json={"general.theme": "dark", "consoles.scrollback": "lots"}))
    show("PUT /settings/secrets/hubs.lab.token {value} (reply: status only)", c.put("/api/v1/settings/secrets/hubs.lab.token", headers=H, json={"value": "tok-SECRET"}),
         lambda b: {k: b[k] for k in ("key","secret","keys","apply","changed")})
    show("GET /settings?key=hubs.lab.token", c.get("/api/v1/settings?key=hubs.lab.token", headers=H), lambda b: {"rows": b["rows"]})
    show("DELETE /settings/tools.openocd", c.delete("/api/v1/settings/tools.openocd", headers=H),
         lambda b: {**b, "rows": [{k: r[k] for k in ("key","value","source","where","shadowed")} for r in b["rows"]]})
    show("POST /settings/test {section: hubs, name: lab} (SET-HUBS not landed)", c.post("/api/v1/settings/test", headers=H, json={"section": "hubs", "name": "lab"}))
    show("GET /settings/schema (2 of the rows)", c.get("/api/v1/settings/schema", headers=H),
         lambda b: {"schema_version": b["schema_version"], "sections": b["sections"], "rows": [r for r in b["rows"] if r["key"] in ("hubs.*.token", "advanced.allowed_hosts")], "n_rows": len(b["rows"])})
    show("GET /api/v1/settings with Host: evil.example -> 403", c.get("/api/v1/settings", headers={**H, "host": "evil.example"}))
print("### settings.changed events on the bus")
for e in events: print(json.dumps(e))
eng.close_all()
```
