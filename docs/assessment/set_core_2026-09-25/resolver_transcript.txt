# The script, then its output. Run: PYTHONPATH=src .venv/bin/python demo.py

```python
import json, os
from pathlib import Path
from harness_manager.settings import Resolver, SecretStore
from harness_manager.settings.secrets import KeyringBackend
from harness_manager.core.errors import HarnessError

root = Path("/tmp/setcore-demo/state"); root.mkdir(parents=True, exist_ok=True)
(root / "update").mkdir(exist_ok=True)
(root / "update" / "settings.json").write_text('{"channel": "beta", "auto": "stage"}')
pol = Path("/tmp/setcore-demo/policy.toml")
pol.write_text('self_update = "notify"\n[lock]\ntools.vivado = "/tools/Xilinx/Vivado/2024.1/bin/vivado"\n'
               '"hubs.*.lease_ttl" = "2h"\n[default]\nupdates.mirrors = ["/lab/mirror"]\n'
               '[hubs.lab]\ntransport = "ssh"\nhost = "mapstone-dev.ecs.soton.ac.uk"\ntoken = "tok-oops"\n')
env = {"HARNESS_MANAGER_OPENOCD": "/env/openocd", "HARNESS_MANAGER_XVC_PORT_BASE": "twelve",
       "HARNESS_MANAGER_KEYRING": "off"}
store = SecretStore(root, keyrings=[KeyringBackend("secret-service", env={})])
r = Resolver.load(root, env=env, policy_path=pol, secrets=store)
def show(k):
    v = r.resolve(k).view()
    extra = {x: v[x] for x in ("shadowed", "capped", "problems", "secret") if v.get(x)}
    print(f"{k:28s} = {json.dumps(v['value']):42s} [{v['source']}{' LOCKED' if v['locked'] else ''}] {v['where']} {extra if extra else ''}")
print("-- policy problems:", r.problems)
r.set("tools.openocd", "/opt/openocd/bin/openocd")
r.set("updates.auto", "stage")
store.set("hubs.lab.token", "tok-SECRET-NEVER-PRINTED")
r = Resolver.load(root, env=env, policy_path=pol, secrets=store)
for k in ("tools.openocd", "tools.vivado", "updates.channel", "updates.auto", "updates.mirrors",
          "updates.check_interval", "debug.xvc_port_base", "hubs.lab.host", "hubs.lab.lease_ttl",
          "hubs.lab.token", "general.theme"):
    show(k)
for k, v in (("tools.vivado", "/mine"), ("hubs.lab.token", "x"), ("debug.port_base", 80)):
    try:
        r.set(k, v)
    except HarnessError as e:
        print(f"set {k}: exit {int(e.code)} {e.message}")
print("-- settings.toml:"); print((root / "settings.toml").read_text())
print("-- update/settings.json:", (root / "update" / "settings.json").read_text().strip())
print("-- secrets/index.json:", (root / "secrets" / "index.json").read_text().replace("\n", " "))
```
